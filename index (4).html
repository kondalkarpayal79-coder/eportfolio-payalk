/* ===== 2. Database demo (localStorage) ===== */
const KEY = "payal_guestbook";
const gList = document.getElementById("g-list");
function loadEntries(){ try { return JSON.parse(localStorage.getItem(KEY)) || []; } catch(e){ return []; } }
function saveEntries(a){ try { localStorage.setItem(KEY, JSON.stringify(a)); } catch(e){} }
function renderEntries(){
  const items = loadEntries();
  gList.innerHTML = "";
  if(!items.length){ gList.innerHTML = "<li>No entries yet. Add the first one above.</li>"; return; }
  items.slice().reverse().forEach(it=>{
    const li = document.createElement("li");
    const b = document.createElement("b"); b.textContent = it.name;
    const t = document.createElement("span"); t.textContent = it.msg;
    li.append(b, t); gList.appendChild(li);
  });
}

  e.preventDefault();
  const name = document.getElementById("g-name").value.trim();
  const msg = document.getElementById("g-msg").value.trim();
  if(!name || !msg) return;
  const a = loadEntries(); a.push({name, msg}); saveEntries(a);
  e.target.reset(); renderEntries();
});
document.getElementById("g-clear").addEventListener("click", ()=>{ saveEntries([]); renderEntries(); });
renderEntries();
